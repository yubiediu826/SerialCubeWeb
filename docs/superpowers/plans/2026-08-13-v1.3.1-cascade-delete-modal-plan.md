# SerialCube v1.3.1 三选项级联 modal — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 替换 v1.2 `NS._cascadeConfirm` browser confirm (3 个调用点: 协议/命令/卡片) 为自定义 3 选项 modal (仅删自己/级联/取消), 引用预览用 chip 分类, 浅/深主题适配.

**Architecture:** 方案 D1+D2+D3+D4 - 1 个统一 modal `dh-cascade-delete-modal` (按 type 参数化), 复用 v1.2.1 `.modal-header-standard` 组件, 新增 `NS._openCascadeModal(type, id, label) → Promise<self|cascade|cancel>`, 替换 3 个调用点, 旧 `_cascadeConfirm` 保留兼容.

**Tech Stack:** HTML + vanilla JS + 内嵌 NS 命名空间; agent-browser 跑 e2e 3 场景 (19-21); bump-version.ps1 + check-readme-sync.ps1 + check-cleanup.ps1 守门.

**Spec:** [`../specs/2026-08-13-v1.3.1-cascade-delete-modal-design.md`](../specs/2026-08-13-v1.3.1-cascade-delete-modal-design.md)

**Design Preview:** [`../../design/v1.3.1-cascade-delete-modal-mockup.html`](../../design/v1.3.1-cascade-delete-modal-mockup.html) (3 场景切换)

NS._cascadeResolve = null;
NS._openCascadeModal = function (type, id, label) {
  return new Promise((resolve) => {
    const refs = NS._findReferences(type, id);
    const orphanCount = (refs.commands ? refs.commands.length : 0)
                      + (refs.cards ? refs.cards.length : 0)
                      + (refs.alerts ? refs.alerts.length : 0);
    NS._renderCascadeModal(type, label, refs, orphanCount);
    NS._cascadeResolve = resolve;
    NS.openModal('dh-cascade-delete');
  });
};

NS._renderCascadeModal = function (type, label, refs, orphanCount) {
  // title + meta
  const titleText = {
    protocol: '确认删除协议 ' + label + '?',
    command: '确认删除命令 ' + label + '?',
    card: '确认删除卡片 ' + label + '?'
  }[type] || '确认删除 ' + label + '?';
  document.getElementById('dh-cascade-delete-title').textContent = titleText;
  // meta (按 type 拉)
  let metaText = '';
  if (type === 'protocol') {
    const proto = (NS.PROTOCOLS || []).find(p => p.id === label.split(' ')[1] || p.id === label);
    metaText = proto ? (proto.name + ' · ' + (proto.commands || []).length + ' 条命令') : label;
  } else if (type === 'command') {
    const proto = (NS.PROTOCOLS || []).find(p => (p.commands || []).some(c => ('0x' + c.id.toString(16).padStart(2, '0').toUpperCase() + ' ' + c.name) === label.replace('确认删除命令 ', '').replace('?', '')));
    metaText = proto ? ('所属协议 ' + proto.id) : label;
  } else {
    const card = (NS.CARDS || []).find(c => label.indexOf(c.title || c.id) >= 0);
    metaText = card ? ('所属协议 ' + card.protocol) : label;
  }
  document.getElementById('dh-cascade-delete-meta').textContent = metaText;
  // 引用预览 chip
  const refsEl = document.getElementById('dh-cascade-delete-refs');
  if (orphanCount === 0) {
    refsEl.innerHTML = '<span class="refs-empty">无引用对象, 仅删自己 = 级联删除</span>';
  } else {
    const chips = [];
    if (refs.commands && refs.commands.length) chips.push('<span class="ref-chip cmd"><span class="num">' + refs.commands.length + '</span> 条命令</span>');
    if (refs.cards && refs.cards.length) chips.push('<span class="ref-chip card"><span class="num">' + refs.cards.length + '</span> 张卡片</span>');
    if (refs.alerts && refs.alerts.length) chips.push('<span class="ref-chip alert"><span class="num">' + refs.alerts.length + '</span> 条告警</span>');
    refsEl.innerHTML = chips.join('');
  }
  // 仅删自己 hint
  const hintEl = document.getElementById('dh-cascade-delete-hint');
  if (orphanCount > 0) {
    const parts = [];
    if (refs.cards && refs.cards.length) parts.push(refs.cards.length + ' 张孤儿卡片');
    if (refs.alerts && refs.alerts.length) parts.push(refs.alerts.length + ' 条孤儿告警');
    if (refs.commands && refs.commands.length) parts.push(refs.commands.length + ' 条孤儿命令');
    document.getElementById('dh-cascade-delete-hint-text').textContent = '选"仅删自己"会留下 ' + parts.join(' + ') + ', 引用悬挂';
    hintEl.style.display = 'flex';
