# SerialCube v1.2 配置中心重构 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 一次性解决 SerialCube 配置中心 5 个 UI/逻辑自洽性问题，引入"激活协议"概念，告警从派生改为独立规则，命令编辑支持三模式复用，为 v1.3 真实模拟预留调试面板占位。

**Architecture:** 方案 D — 单协议聚焦 + 串口绑定。仪表盘只显示 `NS.activeProtoId` 协议的卡片。配置中心 4 tab 全部重做为表格 + 行内操作。命令/告警独立 modal 编辑。删除采用"拒绝 + 列出引用"模式。

**Tech Stack:** HTML + vanilla JS + 内嵌 NS 命名空间（无构建系统）；agent-browser 跑 e2e；bump-version.ps1 + check-readme-sync.ps1 + check-cleanup.ps1 守门。

```js
NS._renderNewCmdFramePreview = function () {
  const s = NS._newCmdState;
  if (!s) return;
  const proto = NS.PROTOCOLS.find((p) => p.id === s.protoId);
  if (!proto) return;
  // 简化的拼帧: HEADER + CMD + dataFields default + CRC
  // 实际应该用 NS.encodeFrame(proto, cmd) 但 v1.2 先简化
  const header = (proto.fields || []).filter((f) => f.type === 'header').map((f) => f.default || '0xAA');
  const cmd = '0x' + (s.id || 0).toString(16).padStart(2, '0').toUpperCase();
  const data = s.dataFields || [];
  // 简化的 hex 拼
  const headerHex = header.join(' ').toUpperCase().replace(/0X/g, '');
  const dataHex = data.map((df) => {
    const v = parseInt(df.default || '0x00', 16) || 0;
    return v.toString(16).padStart(2, '0').toUpperCase();
  }).join(' ');
  const crcHex = '00 00'; // 简化: 真实 CRC 需要 NS.computeCrc
  const html = '<div style="display: flex; gap: 12px; flex-wrap: wrap; align-items: center;">' +
    '<span><span style="color: var(--text-soft); font-size: 10px;">HEADER</span> <span style="color: var(--signal);">' + headerHex + '</span></span>' +
    '<span><span style="color: var(--text-soft); font-size: 10px;">CMD</span> <span style="color: var(--signal);">' + cmd + '</span></span>' +
    '<span><span style="color: var(--text-soft); font-size: 10px;">DATA</span> <span style="color: var(--accent);">' + (dataHex || '—') + '</span></span>' +
    '<span><span style="color: var(--text-soft); font-size: 10px;">CRC</span> <span style="color: var(--danger);">' + crcHex + '</span></span>' +
    '</div>';
  const el = document.getElementById('dh-new-cmd-frame-preview');
  if (el) el.innerHTML = html;
  const sizeEl = document.getElementById('dh-new-cmd-frame-size');
  if (sizeEl) sizeEl.textContent = '共 ' + (header.length + 1 + data.length + 2) + ' 字节';
};
```

```js
NS._findReferences = function (type, id) {
  // type: 'protocol' / 'command' / 'card'
  // id: 协议 id / 命令 id / 卡片 id
  const refs = { commands: [], cards: [], alerts: [] };
  if (type === 'protocol') {
    const proto = NS.PROTOCOLS.find((p) => p.id === id);
    if (proto) refs.commands = proto.commands || [];
    refs.cards = NS.CARDS.filter((c) => c.protocol === id);
    refs.alerts = (NS.ALERTS || []).filter((a) => a.protocol === id);
  } else if (type === 'command') {
    // 找命令所在的协议
    for (const p of NS.PROTOCOLS) {
      const cmd = (p.commands || []).find((c) => c.id === id);
      if (cmd) { refs.protocolId = p.id; refs.protocolName = p.name || p.id; break; }
    }
    refs.cards = NS.CARDS.filter((c) => c.cmd === id);
    refs.alerts = (NS.ALERTS || []).filter((a) => a.cmd === id);
  } else if (type === 'card') {
    refs.alerts = (NS.ALERTS || []).filter((a) => /* 暂用 name 匹配 */ a.name === id);
  }
  return refs;
};
```

```js
NS._cascadeConfirm = function (type, id, label) {
  const refs = NS._findReferences(type, id);
  const parts = [];
  if (refs.commands.length) parts.push(refs.commands.length + ' 条命令');
  if (refs.cards.length) parts.push(refs.cards.length + ' 张卡片');
  if (refs.alerts.length) parts.push(refs.alerts.length + ' 条告警');
  if (parts.length === 0) return 'self';  // 没引用，直接删
  const msg = '确认删除 ' + label + '?\n将被一起删除: ' + parts.join(' / ') + '\n\n[确定] = 级联删除\n[取消] = 不删除';
  return confirm(msg) ? 'cascade' : 'cancel';
};
```

```js
NS.checkAlert = function (v, field, proto) {
  const alerts = (NS.ALERTS || []).filter((a) => a.enabled && a.field === field && a.protocol === proto);
  for (const a of alerts) {
    let triggered = false;
    if (a.op === '>') triggered = v > a.value;
    else if (a.op === '<') triggered = v < a.value;
    else if (a.op === '=') triggered = v === a.value;
    else if (a.op === '!=') triggered = v !== a.value;
    else if (a.op === 'range' && Array.isArray(a.value)) triggered = v < a.value[0] || v > a.value[1];
    if (triggered) return { severity: a.severity, alertId: a.id, name: a.name };
  }
  return null;
};
```
