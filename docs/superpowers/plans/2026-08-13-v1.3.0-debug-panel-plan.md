> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实装 v1.2.0 留的"真实模拟调试面板"占位, 1 人多 tab 模拟设备未到调试场景, BroadcastChannel 主从 + Mutator 单条注入 + 预设场景 chip + Stats 实时计数。

**Architecture:** 方案 D1+D2+D3+D4+D5+D6 - 仪表盘右下角 ⚙ 折叠浮窗 (Lucide settings SVG) + 4 段布局 (Role/Channel/Mutator/Stats), BroadcastChannel 'serialcube-debug-v1' 同源通信, 抽象 `NS._debugTransport` 接口 (BC MVP, 留 hook 升 WebRTC), 设备端 `NS._debugDeviceState` 同步副本 (不污染 currentVals), 复用 `NS.buildFrame` 生成 ack 帧, 预设 chip 从 active protocol dataFields 自动推。

**Tech Stack:** HTML + vanilla JS + 内嵌 NS 命名空间 (无构建系统); BroadcastChannel API (浏览器原生); agent-browser 跑 e2e 6 场景 (10-12 + 浅/深主题截图); bump-version.ps1 + check-readme-sync.ps1 + check-cleanup.ps1 守门。

// v1.3.0: 调试面板 - 设备端接收命令帧 + 生成 ack
NS._recvDebugFrame = function (msg) {
  if (NS._debugRole !== 'device') return;
  // mute 检查
  if (NS._debugMutedUntil > Date.now()) {
    NS.toast('设备端 mute 中, 已丢弃命令', 'info', 1000);
    return;
  }
  // 解析 cmd 帧
  const proto = (NS.PROTOCOLS || []).find(p => p.id === NS.activeProtoId);
  if (!proto) return;
  const cmd = (proto.commands || []).find(c => c.id === msg.cmdId);
  if (!cmd) { NS._debugStats.err++; NS._updateDebugStatsUI(); return; }
  // 找 ack cmd (按 cmd.ackCmdId 关联, 找不到就用下一个同 dir 的 cmd)
  const ackCmd = (proto.commands || []).find(c => c.id === (cmd.ackCmdId || msg.cmdId));
  if (!ackCmd) { NS._debugStats.err++; NS._updateDebugStatsUI(); return; }
  // 临时切换 currentVals 到 device state, buildFrame 用 currentVals 编码
  const _savedVals = { ...NS.currentVals };
  try {
    NS.currentVals = { ...NS._debugDeviceState };
    const ack = NS.buildFrame(proto, ackCmd);
    NS.currentVals = _savedVals;  // 立即还原
    if (ack.error) { NS._debugStats.err++; NS._updateDebugStatsUI(); return; }
    // CRC override
    if (NS._crcOverride === 'force-bad' && ack.bytes.length >= 2) {
      // 倒数第 2 字节 (假设 CRC 在帧尾, v1.2 BMS 协议)
      ack.bytes[ack.bytes.length - 2] = (ack.bytes[ack.bytes.length - 2] ^ 0xFF) & 0xFF;
      ack.bytes[ack.bytes.length - 1] = (ack.bytes[ack.bytes.length - 1] ^ 0xFF) & 0xFF;
      NS._crcOverride = null;  // 一次有效
    }
    // 发 ack
    NS._debugBcTransport.send({
      type: 'ack-frame',
      payload: ack.bytes,
      ackCmdId: ackCmd.id,
      ts: Date.now()
    });
    NS._debugStats.tx++;
    NS._updateDebugStatsUI();
  } catch (e) {
    NS.currentVals = _savedVals;  // 异常也还原
    NS._debugStats.err++;
    NS._updateDebugStatsUI();
    NS.toast('设备端生成 ack 失败: ' + e.message, 'error');
  }
};

// v1.3.0: 调试面板 - Mutator 单条注入 (client 改 currentVals / device 改 device state)
NS._injectMutation = function (fieldName, value) {
  if (!fieldName) return;
  const proto = (NS.PROTOCOLS || []).find(p => p.id === NS.activeProtoId);
  if (!proto) { NS.toast('请先选协议', 'warn'); return; }
  const field = (proto.commands || []).flatMap(c => c.dataFields || []).find(f => f.name === fieldName);
  if (!field) { NS.toast('字段不存在: ' + fieldName, 'error'); return; }
  // 范围校验
  const v = Number(value);
  if (isNaN(v)) { NS.toast('值必须是数字', 'error'); return; }
  if (field.range && (v < field.range[0] || v > field.range[1])) {
    NS.toast(`${fieldName} 超出范围 [${field.range[0]}, ${field.range[1]}]`, 'warn');
    return;
  }
  if (NS._debugRole === 'device') {
    NS._debugDeviceState[fieldName] = v;
    NS.toast(`设备端 ${fieldName} = ${v}`, 'info', 1000);
  } else {
    NS.currentVals[fieldName] = v;
