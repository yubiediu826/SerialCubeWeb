# SerialCube v1.2.1 UI 一致性修复 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 一次性解决 v1.2.0 部署后用户实测反馈的 8 个 UI/UX 问题：4 个 modal 关闭 X 位置错 / 协议编辑内容错位 / "选择协议 modal"冗余 / 仪表盘底部"设置值"常驻 / 协议条"引导"按钮冗余 / "新建协议"按钮重复 3 处 / 帧预览缺失 / 帧预览主题适配错。

```js
/**
 * 渲染帧预览到指定容器。共享函数，命令编辑 + 协议编辑 modal 都用。
 * @param {string} containerId - 目标容器元素 id
 * @param {string} commandId - 命令 ID
 * @returns {void}
 */
NS.renderFramePreview = function (containerId, commandId) {
  const el = document.getElementById(containerId);
  if (!el) {
    console.warn('[renderFramePreview] container not found:', containerId);
    return;
  }

  // 找命令
  const cmd = NS.findCommandById(commandId);
  if (!cmd) {
    el.innerHTML = '<div class="frame-preview-empty">—</div>';
    return;
  }

  // 生成 byte 序列
  let frame;
  try {
    frame = NS.buildFrame(cmd);
  } catch (e) {
    console.error('[renderFramePreview] buildFrame error:', e);
    el.innerHTML = '<div class="frame-preview-empty">—</div>';
    return;
  }

  if (!frame || !frame.bytes || frame.bytes.length === 0) {
    el.innerHTML = '<div class="frame-preview-empty">—</div>';
    return;
  }

  // 渲染 byte 序列（沿用 v1.2 颜色块 + 标注）
  el.innerHTML = '';
  const bytesEl = document.createElement('div');
  bytesEl.className = 'frame-bytes';

  const labels = (frame.labels || []);
  frame.bytes.forEach((b, idx) => {
    const byteEl = document.createElement('span');
    byteEl.className = 'byte ' + (b.type || 'data');
    byteEl.textContent = b.value.toString(16).padStart(2, '0').toUpperCase();
    if (labels[idx]) byteEl.title = labels[idx];
    bytesEl.appendChild(byteEl);
  });

  el.appendChild(bytesEl);

  // 添加 label 标注
  if (labels.length > 0) {
    const labelsEl = document.createElement('div');
    labelsEl.className = 'frame-labels';
    labels.forEach(l => {
      const span = document.createElement('span');
      span.textContent = l;
      labelsEl.appendChild(span);
    });
    el.appendChild(labelsEl);
  }
};
```

/**
 * 协议编辑 modal 的帧预览刷新。dropdown 切换时 + 任一字段/命令改动时调用。

NS.bindProtocolFieldInputs = function () {

- [ ] **Step 5: 加 `NS.openDashboardSettingsModal()` 函数**

在 NS 全局函数区加：

```js
/**
 * 打开"设置值 / 字节预览" modal。
 * 前置条件: 已连接串口 + 已选 activeProtoId (按钮 disabled 态已拦截)
 */
NS.openDashboardSettingsModal = function () {
  if (!NS.serialConnected) {
    alert('请先连接串口');
    return;
  }
  if (!NS.activeProtoId) {
    alert('请先选择协议');
    return;
  }

  const proto = NS.PROTOCOLS.find(p => p.id === NS.activeProtoId);
  if (!proto || !proto.commands || proto.commands.length === 0) {
    alert('当前协议无命令');
    return;
  }

  // 填充命令下拉
  const cmdSelect = document.getElementById('dash-settings-cmd-select');
  cmdSelect.innerHTML = proto.commands.map((c, i) => `
    <option value="${escapeAttr(c.id)}">${i+1}. ${escapeHtml(c.name)} (0x${c.id.toString(16)})</option>
  `).join('');

  // 填充字段输入（取第 1 个命令的字段）
  const firstCmd = proto.commands[0];
  const fieldsEl = document.getElementById('dash-settings-fields');
  fieldsEl.innerHTML = (firstCmd.fields || []).map(f => `
    <label>
      <span>${escapeHtml(f)}</span>
      <input type="text" data-cmd-field="${escapeAttr(f)}" value="0" />
    </label>
  `).join('');

  // 绑定字段输入事件
  fieldsEl.querySelectorAll('input').forEach(input => {
    input.addEventListener('input', () => NS.updateDashboardSettingsPreview());
  });

  cmdSelect.onchange = () => {
    const newCmd = proto.commands.find(c => c.id === cmdSelect.value);
    if (!newCmd) return;
    fieldsEl.innerHTML = (newCmd.fields || []).map(f => `
      <label>
        <span>${escapeHtml(f)}</span>
        <input type="text" data-cmd-field="${escapeAttr(f)}" value="0" />
      </label>
    `).join('');
    fieldsEl.querySelectorAll('input').forEach(input => {
      input.addEventListener('input', () => NS.updateDashboardSettingsPreview());
    });
    NS.updateDashboardSettingsPreview();
  };

  // 副标题
  document.getElementById('dash-settings-subtitle').textContent = `${proto.name} / ${firstCmd.name}`;

  // 显示 modal
  document.getElementById('dash-settings-modal').style.display = 'flex';

  // 首次渲染帧预览
  NS.updateDashboardSettingsPreview();
};

NS.closeDashboardSettingsModal = function () {
  document.getElementById('dash-settings-modal').style.display = 'none';
};

let _dashSettingsPreviewTimer = null;
NS.updateDashboardSettingsPreview = function () {
  clearTimeout(_dashSettingsPreviewTimer);
  _dashSettingsPreviewTimer = setTimeout(() => {
    const cmdSelect = document.getElementById('dash-settings-cmd-select');
    if (!cmdSelect || !cmdSelect.value) return;
    NS.renderFramePreview('dash-frame-preview-modal', cmdSelect.value);
  }, 150);
};
```
